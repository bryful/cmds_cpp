cmake_minimum_required(VERSION 3.24)
project(a2d LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(BIT7Z_USE_NATIVE_STRING ON CACHE BOOL "" FORCE)
set(BIT7Z_PATH_SANITIZATION ON CACHE BOOL "" FORCE)
set(BIT7Z_AUTO_PREFIX_LONG_PATHS ON CACHE BOOL "" FORCE)
set(BIT7Z_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BIT7Z_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(BIT7Z_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(bit7z
  URL https://codeload.github.com/rikyoz/bit7z/zip/refs/tags/v4.1.0
  URL_HASH SHA256=16ca6668153c7823567611f18072a9444df1b6b60ed539a8b1e7272835a5227c)
FetchContent_MakeAvailable(bit7z)
add_executable(a2d a2d.cpp)
target_link_libraries(a2d PRIVATE bit7z)
target_compile_features(a2d PRIVATE cxx_std_20)
target_compile_definitions(a2d PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(bit7z PRIVATE /utf-8)
  target_compile_options(a2d PRIVATE /utf-8 /W4 /permissive-)
endif()
option(A2D_DISABLE_PDB "Disable linker PDB generation for environments with incompatible PDB DLLs" OFF)
if(MSVC AND A2D_DISABLE_PDB)
  target_link_options(a2d PRIVATE /DEBUG:NONE /INCREMENTAL:NO)
endif()
